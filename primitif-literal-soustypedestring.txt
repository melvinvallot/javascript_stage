1. Primitif

Les types de base de TypeScript

let name: string = "Alice"
let age: number = 25
let isActive: boolean = true
let nothing: null = null
let notDefined: undefined = undefined


2. Literal

Une valeur exacte comme type

// Au lieu d'accepter tous les strings...
let direction: "left" | "right" | "up" | "down"

direction = "left"  // ✅
direction = "hello" // ❌ Erreur


3. Sous-type de string

Un type plus précis qui hérite de string

type Email = `${string}@${string}.${string}` // template literal type

let email: Email = "alice@gmail.com"  // ✅
let email2: Email = "pas-un-email"    // ❌ Erreur
