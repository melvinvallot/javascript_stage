6. Keyof

Récupère les clés d'un type sous forme d'union

type User = {
    name: string
    age: number
    email: string
}

type UserKeys = keyof User
// équivaut à : "name" | "age" | "email"

function getProperty(user: User, key: keyof User) {
    return user[key]
}

const user: User = { name: "Alice", age: 25, email: "alice@gmail.com" }

getProperty(user, "name")   // ✅
getProperty(user, "phone")  // ❌ Erreur


7. Typeof

Récupère le type d'une variable existante

const user = {
    name: "Alice",
    age: 25
}

type User = typeof user
// équivaut à : { name: string, age: number }

// Utile pour ne pas répéter le type
const user2: typeof user = {
    name: "Bob",
    age: 30
}


8. As Const

Rend un objet/tableau complètement immuable et littéral

// Sans as const
const colors = ["red", "green", "blue"]
// type : string[]  (trop large)

// Avec as const
const colors = ["red", "green", "blue"] as const
// type : readonly ["red", "green", "blue"]  (exact ✅)

// Exemple concret
const config = {
    host: "localhost",
    port: 3000
} as const

config.port = 4000 // ❌ Erreur, c'est readonly !
